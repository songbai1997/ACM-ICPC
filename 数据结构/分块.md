# 例题
## No1、P2801 教主的魔法
https://www.luogu.com.cn/problem/P2801

## 题意
两种操作：

1、区间 [𝑥,𝑦] 每个数都加上 𝑧；

2、查询区间 [𝑥,𝑦] 内大于等于 z 的数的个数。

## 解题思路
我们要询问一个块内大于等于一个数的数的个数，所以需要一个 t 数组对块内排序，a 为原来的（未被排序的）数组。对于整块的修改，使用类似于标记永久化的方式，用 add 数组记录现在块内整体加上的值。
```cpp
#include <bits/stdc++.h>

using namespace std;

const int N = 1000005;
int n, block_num, block_size, L[N], R[N], belong[N], a[N], t[N], add[N];

void build()
{
    block_size = sqrt(n);
    block_num = n / block_size;
    for (int i = 1; i <= block_num; i++) {
        L[i] = (i - 1) * block_size + 1;
        R[i] = i * block_size;
    }
    R[block_num] = n;
    for (int i = 1; i <= block_num; i++)
        for (int j = L[i]; j <= R[i]; j++)
            belong[j] = i;
    for (int i = 1; i <= n; i++) t[i] = a[i];
    for (int i = 1; i <= block_num; i++) sort(t + L[i], t + R[i] + 1);
}

void sort_block(int k)
{
    for (int i = L[k]; i <= R[k]; i++) t[i] = a[i];
    sort(t + L[k], t + R[k] + 1);
}

void update(int l, int r, int d)
{
    int p = belong[l], y = belong[r];
    if (p == y) {
        for (int i = l; i <= r; i++) a[i] += d;
        sort_block(p);
    } else {
        for (int i = l; i <= R[p]; i++) a[i] += d;
        for (int i = p + 1; i < y; i++) add[i] += d;
        for (int i = L[y]; i <= r; i++) a[i] += d;
        sort_block(p);
        sort_block(y);
    }    
}

int query(int l, int r, int c)
{
    int ans = 0;
    int p = belong[l], y = belong[r];
    if (p == y) {
        for (int i = l; i <= r; i++) if (a[i] + add[p] >= c) ans++;
    } else {
        for (int i = l; i <= R[p]; i++) if (a[i] + add[p] >= c) ans++;
        for (int i = p + 1; i < y; i++) ans += t + R[i] + 1 - lower_bound(t + L[i], t + R[i] + 1, c - add[i]);
        for (int i = L[y]; i <= r; i++) if (a[i] + add[y] >= c) ans++;
    }
    return ans;
}

int main()
{
    int q;
    scanf("%d %d", &n, &q);
    for (int i = 1; i <= n; i++) scanf("%d", &a[i]);
    build();
    while (q--) {
        char op;
        int l, r, v;
        scanf(" %c %d %d %d", &op, &l, &r, &v);
        if (op == 'M') {
            update(l, r, v);
        } else {
            printf("%d\n", query(l, r, v));
        }
    }
    return 0;
}
```

## No2、P4168 [Violet] 蒲公英
https://www.luogu.com.cn/problem/P4168

## 题目描述

在乡下的小路旁种着许多蒲公英，而我们的问题正是与这些蒲公英有关。

为了简化起见，我们把所有的蒲公英看成一个长度为 $n$ 的序列 $\{a_1,a_2..a_n\}$，其中 $a_i$ 为一个正整数，表示第 $i$ 棵蒲公英的种类编号。

而每次询问一个区间 $[l, r]$，你需要回答区间里出现次数最多的是哪种蒲公英，如果有若干种蒲公英出现次数相同，则输出种类编号最小的那个。

**注意，你的算法必须是在线的**。

## 输入格式

第一行有两个整数，分别表示蒲公英的数量 $n$ 和询问次数 $m$。

第二行有 $n$ 个整数，第 $i$ 个整数表示第 $i$ 棵蒲公英的种类 $a_i$。

接下来 $m$ 行，每行两个整数 $l_0, r_0$，表示一次询问。输入是加密的，解密方法如下：

令上次询问的结果为 $x$（如果这是第一次询问，则 $x = 0$），设 $l=((l_0+x-1)\bmod n) + 1,r=((r_0+x-1) \bmod n) + 1$。如果 $l > r$，则交换 $l, r$。  
最终的询问区间为计算后的 $[l, r]$。

## 输出格式

对于每次询问，输出一行一个整数表示答案。

## 输入输出样例 #1

### 输入 #1

```
6 3 
1 2 3 2 1 2 
1 5 
3 6 
1 5
```

### 输出 #1

```
1 
2 
1
```

## 说明/提示

#### 数据规模与约定

- 对于 $20\%$ 的数据，保证 $n,m \le 3000$。
- 对于 $100\%$ 的数据，保证 $1\le n \le 40000$，$1\le m \le 50000$，$1\le a_i \le 10^9$，$1 \leq l_0, r_0 \leq n$。

## 解题思路
先进行分块，然后预处理任意两个块范围内的最小众数，对于查询在左右边界的元素，利用二分求其在查询范围内出现的次数。

```cpp
#include <bits/stdc++.h>

using namespace std;

const int N = 4e4 + 10;
const int BLOCK_NUM = 2e2 + 10;
int n, block_num, block_size, L[N], R[N], belong[N], a[N], id2value[N];
pair<int, int> f[BLOCK_NUM][BLOCK_NUM];

void build()
{
    block_size = sqrt(n);
    block_num = n / block_size;
    for (int i = 1; i <= block_num; i++) {
        L[i] = (i - 1) * block_size + 1;
        R[i] = i * block_size;
    }
    R[block_num] = n;
    for (int i = 1; i <= block_num; i++)
        for (int j = L[i]; j <= R[i]; j++)
            belong[j] = i;    
}

int main()
{
    int q;
    scanf("%d %d", &n, &q);
    unordered_map<int, int> id;
    int k = 0;
    for (int i = 1; i <= n; i++) {
        scanf("%d", &a[i]);
        if (id.find(a[i]) == id.end()) id[a[i]] = k++;
        id2value[id[a[i]]] = a[i];
        a[i] = id[a[i]];
    }
    build();
    vector<vector<int>> idx(k);
    for (int i = 1; i <= n; i++) idx[a[i]].push_back(i);

    auto query_cnt = [&](int v, int l, int r) {
        return upper_bound(idx[v].begin(), idx[v].end(), r) - lower_bound(idx[v].begin(), idx[v].end(), l);
    };

    auto update_res = [&](int& mx_v, int& mx_cnt, int v, int cnt) {
        if (cnt > mx_cnt || cnt == mx_cnt && v < mx_v) {
            mx_v = v;
            mx_cnt = cnt;
        }
        return 0;
    };

    auto pre_cal = [&]() {
        for (int i = 1; i <= block_num; i++) {
            vector<int> cnt(k, 0);
            int mx_v = 0, mx_cnt = 0;
            for (int j = i; j <= block_num; j++) {
                for (int k = L[j]; k <= R[j]; k++) {
                    update_res(mx_v, mx_cnt, id2value[a[k]], ++cnt[a[k]]);
                }
                f[i][j] = {mx_v, mx_cnt};
            }
        }
        return 0;
    }();

    auto query = [&](int l, int r) {
        int mx_v = 0, mx_cnt = 0;
        int x = belong[l], y = belong[r];
        if (x == y) {
            for (int i = l; i <= r; i++) update_res(mx_v, mx_cnt, id2value[a[i]], query_cnt(a[i], l, r));
        } else {
            for (int i = l; i <= R[x]; i++) update_res(mx_v, mx_cnt, id2value[a[i]], query_cnt(a[i], l, r));
            for (int i = L[y]; i <= r; i++) update_res(mx_v, mx_cnt, id2value[a[i]], query_cnt(a[i], l, r));
            update_res(mx_v, mx_cnt, f[x + 1][y - 1].first, f[x + 1][y - 1].second);
        }
        return mx_v;
    };

    int v = 0;
    while (q--) {
        int l, r;
        scanf("%d %d", &l, &r);
        l = (l + v - 1) % n + 1;
        r = (r + v - 1) % n + 1;
        if (l > r) swap(l, r);
        v = query(l, r);
        printf("%d\n", v);
    }
    return 0;
}
```
