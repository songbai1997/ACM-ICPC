# 【模板】线段树：单点修改+区间查询
- 模板来源 https://leetcode.cn/circle/discuss/mOr1u6/
- 线段树有两个下标，一个是线段树节点的下标，另一个是线段树维护的区间的下标
- 节点的下标：从 1 开始，如果你想改成从 0 开始，需要把左右儿子下标分别改成 node*2+1 和 node*2+2
- 区间的下标：从 0 开始

```cpp
class SegmentTree {
public:
    #define lson rt << 1
    #define rson rt << 1 | 1
    #define ls lson, l, m
    #define rs rson, m+1, r
    using T = int;

    int n;
    vector<T> tree;

    void pushup(int rt) {
        tree[rt] = max(tree[lson], tree[rson]);
    }

    void build(const vector<T>& a, int rt, int l, int r) {
        if (l == r) {
            tree[rt] = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, ls);
        build(a, rs);
        pushup(rt);
    }

    void update(int rt, int l, int r, int i, T val) {
        if (l == r) {
            tree[rt] = val;
            return;
        }
        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    int queryFirstPos(int rt, int l, int r, int ql, int qr, int target) const {
        if (tree[rt] < target) return -1;
        if (l == r) return l;
        int m = (l + r) / 2;
        int idx = -1;
        if (ql <= m) idx = queryFirstPos(ls, ql, qr, target);
        if (idx != -1) return idx;
        if (qr > m) idx = queryFirstPos(rs, ql, qr, target);
        return idx;
    }

    SegmentTree(int n, T init_val) : SegmentTree(vector<T>(n, init_val)) {}

    SegmentTree(const vector<T>& a) : n(a.size()), tree(2 << bit_width(a.size() - 1)) {
        build(a, 1, 0, n - 1);
    }
};
```

## No1、3479. 水果成篮 III
https://leetcode.cn/problems/fruits-into-baskets-iii/description/

## 题意
给你两个长度为 n 的整数数组，fruits 和 baskets，其中 fruits[i] 表示第 i 种水果的 数量，baskets[j] 表示第 j 个篮子的 容量。

你需要对 fruits 数组从左到右按照以下规则放置水果：

每种水果必须放入第一个 容量大于等于 该水果数量的 最左侧可用篮子 中。
每个篮子只能装 一种 水果。
如果一种水果 无法放入 任何篮子，它将保持 未放置。
返回所有可能分配完成后，剩余未放置的水果种类的数量。

## 解题思路
线段树维护区间最大值+二分答案
```cpp
class SegmentTree {
public:

            return;

        build(a, rs);

            tree[rt] = val;
            return;

        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    int queryFirstPos(int rt, int l, int r, int ql, int qr, int target) const {
        if (tree[rt] < target) return -1;
        if (l == r) return l;
        int m = (l + r) / 2;
        int idx = -1;
        if (ql <= m) idx = queryFirstPos(ls, ql, qr, target);
        if (idx != -1) return idx;
        if (qr > m) idx = queryFirstPos(rs, ql, qr, target);
        return idx;

class Solution {
public:
    int numOfUnplacedFruits(vector<int>& fruits, vector<int>& baskets) {
        SegmentTree t(baskets);
        int ans = 0;
        for (int x : fruits) {
            int idx = t.queryFirstPos(1, 0, baskets.size() - 1, 0, baskets.size() - 1, x);
            if (idx == -1) {
                ans++;
            } else {
                t.update(1, 0, baskets.size() - 1, idx, -1);
            }
        }
        return ans;
    }
};
```

## No2、2286. 以组为单位订音乐会的门票
https://leetcode.cn/problems/booking-concert-tickets-in-groups/description/

## 题意
一个音乐会总共有 n 排座位，编号从 0 到 n - 1 ，每一排有 m 个座椅，编号为 0 到 m - 1 。你需要设计一个买票系统，针对以下情况进行座位安排：

- 同一组的 k 位观众坐在 同一排座位，且座位连续 。
- k 位观众中 每一位 都有座位坐，但他们 不一定 坐在一起。

由于观众非常挑剔，所以：

- 只有当一个组里所有成员座位的排数都 小于等于 maxRow ，这个组才能订座位。每一组的 maxRow 可能 不同 。
- 如果有多排座位可以选择，优先选择 最小 的排数。如果同一排中有多个座位可以坐，优先选择号码 最小 的。

请你实现 BookMyShow 类：

- BookMyShow(int n, int m) ，初始化对象，n 是排数，m 是每一排的座位数。
- int[] gather(int k, int maxRow) 返回长度为 2 的数组，表示 k 个成员中 第一个座位 的排数和座位编号，这 k 位成员必须坐在 同一排座位，且座位连续 。换言之，返回最小可能的 r 和 c 满足第 r 排中 [c, c + k - 1] 的座位都是空的，且 r <= maxRow 。如果 无法 安排座位，返回 [] 。
- boolean scatter(int k, int maxRow) 如果组里所有 k 个成员 不一定 要坐在一起的前提下，都能在第 0 排到第 maxRow 排之间找到座位，那么请返回 true 。这种情况下，每个成员都优先找排数 最小 ，然后是座位编号最小的座位。如果不能安排所有 k 个成员的座位，请返回 false 。

## 解题思路
线段树维护区间最大值和区间和。实现`gather`功能时二分第一个满足要求的行；实现`scatter`功能时，先查询区间和判断是否可行，若可行再查找可行的行。

```cpp
class SegmentTree {
public:
    #define lson rt << 1
    #define rson rt << 1 | 1
    #define ls lson, l, m
    #define rs rson, m+1, r
    using T = int;
    using T2 = long long;

    using Node = struct {
        T mx;
        T2 sum;
    };

    int n;
    vector<Node> tree;

    void pushup(int rt) {
        tree[rt].mx = max(tree[lson].mx, tree[rson].mx);
        tree[rt].sum = tree[lson].sum + tree[rson].sum;
    }

    void build(const vector<T>& a, int rt, int l, int r) {
        if (l == r) {
            tree[rt].mx = a[l];
            tree[rt].sum = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, ls);
        build(a, rs);
        pushup(rt);
    }

    void update(int rt, int l, int r, int i, T val) {
        if (l == r) {
            tree[rt].sum += val;
            tree[rt].mx = tree[rt].sum;
            return;
        }
        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    T2 querySum(int rt, int l, int r, int ql, int qr) const {
        if (ql <= l && r <= qr) {
            return tree[rt].sum;
        }
        int m = (l + r) / 2;
        T2 sum = 0;
        if (ql <= m) sum = querySum(ls, ql, qr);
        if (qr > m) sum += querySum(rs, ql, qr);
        return sum;
    }

    int queryFirstPos(int rt, int l, int r, int ql, int qr, int target) const {
        if (tree[rt].mx < target) return -1;
        if (l == r) return l;
        int m = (l + r) / 2;
        int idx = -1;
        if (ql <= m) idx = queryFirstPos(ls, ql, qr, target);
        if (idx != -1) return idx;
        if (qr > m) idx = queryFirstPos(rs, ql, qr, target);
        return idx;
    }

    SegmentTree(int n, T init_val) : SegmentTree(vector<T>(n, init_val)) {}

    SegmentTree(const vector<T>& a) : n(a.size()), tree(2 << bit_width(a.size() - 1)) {
        build(a, 1, 0, n - 1);
    }
};

class BookMyShow {
public:
    SegmentTree *t;
    int n, m;

    BookMyShow(int n, int m): n(n), m(m) {
        t = new SegmentTree(n, m);
    }
    
    vector<int> gather(int k, int maxRow) {
        int r = t->queryFirstPos(1, 0, n - 1, 0, maxRow, k);
        if (r == -1) return {};
        int cnt = t->querySum(1, 0, n - 1, r, r);
        t->update(1, 0, n - 1, r, -k);
        return {r, m - cnt};
    }
    
    bool scatter(int k, int maxRow) {
        long long sum = t->querySum(1, 0, n - 1, 0, maxRow);
        if (sum < k) return false;
        int r = t->queryFirstPos(1, 0, n - 1, 0, maxRow, 1);
        while (k > 0) {
            int cnt = t->querySum(1, 0, n - 1, r, r);
            int delta = min(k, cnt);
            t->update(1, 0, n - 1, r, -delta);
            k -= delta;
            r++;
        }
        return true;
    }
};
```
## No3、3161. 物块放置查询
https://leetcode.cn/problems/block-placement-queries/description/

## 题意
有一条无限长的数轴，原点在 0 处，沿着 x 轴 正 方向无限延伸。

给你一个二维数组 queries ，它包含两种操作：

- 操作类型 1 ：queries[i] = [1, x] 。在距离原点 x 处建一个障碍物。数据保证当操作执行的时候，位置 x 处 没有 任何障碍物。
- 操作类型 2 ：queries[i] = [2, x, sz] 。判断在数轴范围 [0, x] 内是否可以放置一个长度为 sz 的物块，这个物块需要 完全 放置在范围 [0, x] 内。如果物块与任何障碍物有重合，那么这个物块 不能 被放置，但物块可以与障碍物刚好接触。注意，你只是进行查询，并 不是 真的放置这个物块。每个查询都是相互独立的。

请你返回一个 boolean 数组results ，如果第 i 个操作类型 2 的操作你可以放置物块，那么 results[i] 为 true ，否则为 false 。

## 解题思路
维护一个区间包含左右端点的结果。

```cpp
class SegmentTree {
public:
    #define lson rt << 1
    #define rson rt << 1 | 1
    #define ls lson, l, m
    #define rs rson, m+1, r
    using T = int;

    using Node = struct {
        T sum;
        T v;
        T lv;
        T rv;
    };

    int n;
    vector<Node> t;

    void merge(Node& rt, Node& l, Node& r) {
        rt.sum = l.sum + r.sum;
        rt.v = max(max(l.v, r.v), l.rv + r.lv);
        rt.lv = l.lv + (l.lv == l.sum ? r.lv : 0);
        rt.rv = r.rv + (r.rv == r.sum ? l.rv : 0);
    }

    void pushup(int rt) {
        merge(t[rt], t[lson], t[rson]);
    }

    void build(const vector<T>& a, int rt, int l, int r) {
        if (l == r) {
            t[rt].sum = t[rt].v = t[rt].lv = t[rt].rv = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, ls);
        build(a, rs);
        pushup(rt);
    }

    void update(int rt, int l, int r, int i, T val) {
        if (l == r) {
            t[rt].v = t[rt].lv = t[rt].rv = val;
            return;
        }
        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    Node querySum(int rt, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) {
            return t[rt];
        }
        int m = (l + r) / 2;
        if (qr <= m) return querySum(ls, ql, qr);
        if (ql > m) return querySum(rs, ql, qr);
        Node left = querySum(ls, ql, qr);
        Node right = querySum(rs, ql, qr);
        Node node = { 0 };
        merge(node, left, right);
        return node;
    }

    bool query(int ql, int qr, int sz) {
        Node node = querySum(1, 0, n - 1, ql, qr);
        return node.v + 1 >= sz;
    }

    SegmentTree(int n, T init_val) : SegmentTree(vector<T>(n, init_val)) {}

    SegmentTree(const vector<T>& a) : n(a.size()), t(2 << bit_width(a.size() - 1)) {
        build(a, 1, 0, n - 1);
    }
};

class Solution {
public:
    vector<bool> getResults(vector<vector<int>>& queries) {
        int n = 0;
        for (auto& q : queries) n = max(n, q[1]);
        SegmentTree t(n + 1, 1);
        t.update(1, 0, n, 0, 0);
        vector<bool> ans;
        for (auto& q : queries) {
            if (q[0] == 1) {
                t.update(1, 0, n, q[1], 0);
            } else {
                ans.push_back(t.query(0, q[1] - 1, q[2]));
            }
        }
        return ans;
    }
};
```

## No4、2213. 由单个字符重复的最长子字符串
https://leetcode.cn/problems/longest-substring-of-one-repeating-character/description/

## 题意
给你一个下标从 0 开始的字符串 s 。另给你一个下标从 0 开始、长度为 k 的字符串 queryCharacters ，一个下标从 0 开始、长度也是 k 的整数 下标 数组 queryIndices ，这两个都用来描述 k 个查询。

第 i 个查询会将 s 中位于下标 queryIndices[i] 的字符更新为 queryCharacters[i] 。

返回一个长度为 k 的数组 lengths ，其中 lengths[i] 是在执行第 i 个查询 之后 s 中仅由 单个字符重复 组成的 最长子字符串 的 长度 。

## 解题思路
维护如下区间信息：
- 此区间是否均是相同字符
- 此区间由单个字符重复组成的最长子字符串的长度
- 此区间以左端点开始的由单个字符重复组成的最长子字符串的长度
- 此区间以右端点结束的由单个字符重复组成的最长子字符串的长度
- 此区间左端点字符
- 此区间右端点字符

```cpp
class SegmentTree {
public:
    #define lson rt << 1
    #define rson rt << 1 | 1
    #define ls lson, l, m
    #define rs rson, m+1, r

    using Node = struct {
        bool all;
        int v;
        int lv;
        int rv;
        int lc;
        int rc;
    };

    int n;
    vector<Node> t;

    void merge(Node& rt, Node& l, Node& r) {
        rt.all = l.all && r.all && l.lc == r.lc;
        rt.v = max(max(l.v, r.v), l.rv + (l.rc == r.lc ? r.lv : 0));
        rt.lv = l.lv + (l.all && l.lc == r.lc ? r.lv : 0);
        rt.rv = r.rv + (r.all && r.rc == l.rc ? l.rv : 0);
        rt.lc = l.lc;
        rt.rc = r.rc;
    }

    void pushup(int rt) {
        merge(t[rt], t[lson], t[rson]);
    }

    void build(const string& a, int rt, int l, int r) {
        if (l == r) {
            t[rt].all = true;
            t[rt].v = t[rt].lv = t[rt].rv = 1;
            t[rt].lc = t[rt].rc = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, ls);
        build(a, rs);
        pushup(rt);
    }

    void update(int rt, int l, int r, int i, int val) {
        if (l == r) {
            t[rt].all = true;
            t[rt].v = t[rt].lv = t[rt].rv = 1;
            t[rt].lc = t[rt].rc = val;
            return;
        }
        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    Node querySum(int rt, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) {
            return t[rt];
        }
        int m = (l + r) / 2;
        if (qr <= m) return querySum(ls, ql, qr);
        if (ql > m) return querySum(rs, ql, qr);
        Node left = querySum(ls, ql, qr);
        Node right = querySum(rs, ql, qr);
        Node node = { 0 };
        merge(node, left, right);
        return node;
    }

    int query(int ql, int qr) {
        Node node = querySum(1, 0, n - 1, ql, qr);
        return node.v;
    }

    SegmentTree(string& a) : n(a.size()), t(2 << bit_width(a.size() - 1)) {
        build(a, 1, 0, n - 1);
    }
};

class Solution {
public:
    vector<int> longestRepeating(string s, string queryCharacters, vector<int>& queryIndices) {
        SegmentTree t(s);
        int k = queryCharacters.size();
        vector<int> ans(k, 0);
        for (int i = 0; i < k; i++) {
            t.update(1, 0, s.size() - 1, queryIndices[i], queryCharacters[i]);
            ans[i] = t.query(0, s.size() - 1);
        }
        return ans;
    }
};
```

## No5、3165. 不包含相邻元素的子序列的最大和
https://leetcode.cn/problems/maximum-sum-of-subsequence-with-non-adjacent-elements/description/

## 题意
给你一个整数数组 nums 和一个二维数组 queries，其中 queries[i] = [posi, xi]。

对于每个查询 i，首先将 nums[posi] 设置为 xi，然后计算查询 i 的答案，该答案为 nums 中 不包含相邻元素 的 子序列 的 最大 和。

返回所有查询的答案之和。

由于最终答案可能非常大，返回其对 109 + 7 取余 的结果。

子序列 是指从另一个数组中删除一些或不删除元素而不改变剩余元素顺序得到的数组。

## 解题思路
维护如下区间信息：
- 此区间的答案
- 此区间不包含左端点的答案
- 此区间不包含右端点的答案
- 此区间不包含左右端点的答案

```cpp
    }

class Solution {
public:
    int maximumSumSubsequence(vector<int>& nums, vector<vector<int>>& queries) {
        SegmentTree t(nums);
        int n = nums.size();
        int mod = 1e9 + 7;
        long long ans = 0;
        for (auto& q : queries) {
            t.update(1, 0, n - 1, q[0], q[1]);
            long long value = t.query(0, n - 1);
            ans = (ans + value) % mod;
        }
        return ans;
    }
};
```

## No6、3410. 删除所有值为某个元素后的最大子数组和
https://leetcode.cn/problems/maximize-subarray-sum-after-removing-all-occurrences-of-one-element/description/

## 题意
给你一个整数数组 nums 。

你可以对数组执行以下操作 至多 一次：

- 选择 nums 中存在的 任意 整数 X ，确保删除所有值为 X 的元素后剩下数组 非空 。
- 将数组中 所有 值为 X 的元素都删除。

请你返回 所有 可能得到的数组中 最大 子数组 和为多少。

## 解题思路
先记录每个值的下标，删除操作就是将对应位置设置为0。需要注意的点是，当查询出的最优结果是0时不能更新答案（由删除的元素形成的结果）。
查询操作为求区间最大子数组和，维护如下区间信息：
- 此区间的答案
- 此区间包含左端点的答案
- 此区间包含右端点的答案
- 此区间包含所有元素的答案
```cpp
class SegmentTree {
public:
    #define lson rt << 1
    #define rson rt << 1 | 1
    #define ls lson, l, m
    #define rs rson, m+1, r
    
    using T = long long;
    using Node = struct {
        T v;
        T lv;
        T rv;
        T av;
    };

    int n;
    vector<Node> t;

    void merge(Node& rt, Node& l, Node& r) {
        rt.v = max(max(l.v, r.v), l.rv + r.lv);
        rt.lv = max(l.lv, l.av + r.lv);
        rt.rv = max(r.rv, r.av + l.rv);
        rt.av = l.av + r.av; 
    }

    void pushup(int rt) {
        merge(t[rt], t[lson], t[rson]);
    }

    void build(vector<int>& a, int rt, int l, int r) {
        if (l == r) {
            t[rt].v = t[rt].lv = t[rt].rv = t[rt].av = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, ls);
        build(a, rs);
        pushup(rt);
    }

    void update(int rt, int l, int r, int i, int val) {
        if (l == r) {
            t[rt].v = t[rt].lv = t[rt].rv = t[rt].av = val;
            return;
        }
        int m = (l + r) / 2;
        if (i <= m) {
            update(ls, i, val);
        } else {
            update(rs, i, val);
        }
        pushup(rt);
    }

    Node querySum(int rt, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) {
            return t[rt];
        }
        int m = (l + r) / 2;
        if (qr <= m) return querySum(ls, ql, qr);
        if (ql > m) return querySum(rs, ql, qr);
        Node left = querySum(ls, ql, qr);
        Node right = querySum(rs, ql, qr);
        Node node = { 0 };
        merge(node, left, right);
        return node;
    }

    T query(int ql, int qr) {
        Node node = querySum(1, 0, n - 1, ql, qr);
        return node.v;
    }

    SegmentTree(vector<int>& a) : n(a.size()), t(2 << bit_width(a.size() - 1)) {
        build(a, 1, 0, n - 1);
    }
};

class Solution {
public:
    long long maxSubarraySum(vector<int>& nums) {
        unordered_map<int, vector<int>> idx;
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            idx[nums[i]].push_back(i);
        }
        SegmentTree t(nums);
        long long ans = t.query(0, n - 1);
        for (auto& [key, pos] : idx) {
            for (int& p : pos) {
                t.update(1, 0, n - 1, p, 0);
            }
            long long val = t.query(0, n - 1);
            if (val != 0) ans = max(ans, val);
            for (int& p : pos) {
                t.update(1, 0, n - 1, p, key);
            }
        }
        return ans;
    }
};
```
